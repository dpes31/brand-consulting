function scaleSlides() {
    const wrappers = document.querySelectorAll('.slide-wrapper');
    const availableWidth = window.innerWidth - 300;
    const baseWidth = 1122;
    let scale = availableWidth / baseWidth;
    if (scale > 1) scale = 1;
    wrappers.forEach(el => {
        el.style.transform = `scale(${scale})`;
        const h = el.getBoundingClientRect().height / scale;
        el.style.marginBottom = `-${h * (1 - scale) - 40}px`;
    });
}
window.addEventListener('beforeprint', function() {
    document.querySelectorAll('.slide-wrapper').forEach(function(el) {
        el._savedTransform = el.style.transform;
        el._savedMarginBottom = el.style.marginBottom;
        el.style.transform = '';
        el.style.marginBottom = '';
    });
});
window.addEventListener('afterprint', function() {
    document.querySelectorAll('.slide-wrapper').forEach(function(el) {
        el.style.transform = el._savedTransform || '';
        el.style.marginBottom = el._savedMarginBottom || '';
    });
});
window.addEventListener('resize', scaleSlides);
window.addEventListener('load', scaleSlides);

document.addEventListener('DOMContentLoaded', function() {
    var sections = document.querySelectorAll('.slide-wrapper');
    var navItems = document.querySelectorAll('.nav-item');
    var navObserver = new IntersectionObserver(function(entries) {
        entries.forEach(function(entry) {
            if (entry.isIntersecting) {
                var id = entry.target.getAttribute('id').replace('wrap-', '');
                navItems.forEach(function(item) {
                    item.classList.remove('active');
                    if (item.getAttribute('href') === '#' + id) item.classList.add('active');
                });
            }
        });
    }, { threshold: 0.3 });
    sections.forEach(function(section) { navObserver.observe(section); });
});
